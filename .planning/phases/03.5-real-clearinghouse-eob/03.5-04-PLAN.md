```typescript
import type { IClearinghouseEOBSource } from './source-types';
import type { RawEOBDocument } from '../types';
import { PayerConnectionError } from '../types';
import type { DXCCredentialsBlob } from './dentalxchange';
import { parse835 } from './edi835-parser';
import { mapParsed835ToRawEOB } from './edi835-mappers';
import { getConfig } from '../../../config';

export interface RealSourceConfig {
  payerName: string;
  env: 'sandbox' | 'production';
  credentials: DXCCredentialsBlob | null;
}

const TIMEOUT_MS = 30000;
const MAX_RETRIES = 3;
const RETRY_BASE_MS = 1000;
const PAGE_SIZE = 100;
let lastRequestAt = 0; // module-scope throttle state
const MIN_REQUEST_INTERVAL_MS = 0; // adjust if DXC publishes a rate limit

interface ERAReference { eraId: string; receivedDate: string; payerName: string; }

export class RealClearinghouseEOBSource implements IClearinghouseEOBSource {
  private readonly baseUrl: string;
  private readonly apiKey: string;

  constructor(private readonly cfg: RealSourceConfig) {
    const config = getConfig();
    const url = cfg.env === 'sandbox' ? config.dentalXChange?.sandboxApiBaseUrl : config.dentalXChange?.paymentApiBaseUrl;
    if (!url) throw new PayerConnectionError(`DXC ${cfg.env} base URL not configured`);
    this.baseUrl = url.replace(/\/$/, '');
    const key = cfg.credentials?.payment?.apiKey ?? cfg.credentials?.eligibility?.apiKey;
    if (!key || key === 'placeholder') throw new PayerConnectionError(`DXC payment API key missing for ${cfg.payerName}`);
    this.apiKey = key;
  }

  async fetch(params: { dateFrom: string; dateTo: string; clinicNpi: string }): Promise<RawEOBDocument[]> {
    const eraRefs = await this.listERAs(params);
    const documents: RawEOBDocument[] = [];
    for (const ref of eraRefs) {
      try {
        const rawEdi = await this.fetchERA(ref.eraId);
        const parsed = await parse835(rawEdi);
        const doc = mapParsed835ToRawEOB(parsed);
        documents.push(doc);
      } catch (err) {
        const safe = err instanceof Error ? err.message.replace(/\d{8,}/g, '[REDACTED]') : 'unknown';
        console.error(`[DXC] Failed to retrieve ERA ${ref.eraId} for ${this.cfg.payerName}: ${safe}`);
      }
    }
    return documents;
  }

  // listERAs: pagination loop matching OpenDental client.ts:104-128
  // fetchERA: single GET that returns raw 835 EDI text (Accept: application/edi-x12 or text/plain)
  // request<T>: 3-retry loop with 30s AbortController timeout, 5xx-only retry, exponential backoff,
  //   PHI-safe error envelope. Verbatim copy of client.ts:33-90 with PMSConnectionError → PayerConnectionError
  //   and the response.text() body STRIPPED from error messages.
}
```

```typescript
import type { Job } from '../../../db/schema';
import { createJob, updateJobStatus } from '../manager';
import { enumerateEobConfigs, syncEobsForConfig, autoPostEligibleEobs } from '../../eob/engine';

/**
 * Parent eob_sync handler: enumerates enabled payer configs, enqueues one
 * eob_sync_child per config, completes immediately. Per RESEARCH.md Pitfall 2 option B.
 */
export async function eobSyncHandler(job: Job): Promise<void> {
  const configs = await enumerateEobConfigs(job.clinicId);
  let enqueued = 0;
  for (const config of configs) {
    await createJob({
      clinicId: job.clinicId,
      jobType: 'eob_sync_child',
      priority: job.priority ?? 5,
      triggeredBy: job.triggeredBy,
      triggerSource: 'scheduler',
      relatedEntityType: 'payer_config',
      relatedEntityId: config.id,
    });
    enqueued++;
  }
  await updateJobStatus(job.id, 'completed', {
    completedAt: new Date(),
    processedItems: enqueued,
    result: { enqueuedChildren: enqueued, message: `Enqueued ${enqueued} eob_sync_child jobs` },
  });
}

/**
 * Child handler: processes ONE (clinic, payerConfig, dateRange) tuple.
 * Idempotency relies on existing (clinicId, checkNumber, payerName) uniqueness in engine.
 */
export async function eobSyncChildHandler(job: Job): Promise<void> {
  if (!job.relatedEntityId) {
    await updateJobStatus(job.id, 'completed', {
      completedAt: new Date(),
      result: { error: 'No payer_config id in relatedEntityId' },
    });
    return;
  }
  const dateFrom = new Date(Date.now() - 30 * 86400000).toISOString().split('T')[0];
  const dateTo = new Date().toISOString().split('T')[0];
  const result = await syncEobsForConfig(job.clinicId, job.relatedEntityId, dateFrom, dateTo);
  const posted = await autoPostEligibleEobs(job.clinicId);
  await updateJobStatus(job.id, 'completed', {
    completedAt: new Date(),
    processedItems: result.totalRetrieved,
    failedItems: result.errors,
    result: { ...result, autoPostedToPms: posted },
  });
}
```
