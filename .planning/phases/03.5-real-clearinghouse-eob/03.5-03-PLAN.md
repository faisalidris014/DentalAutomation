function parseCompositeProcedureCode(svc01: string): { qualifier: string; code: string } {
  // SVC01 is "QUAL:CODE" e.g. "AD:D2740" — split on first colon
  const idx = svc01.indexOf(':');
  if (idx < 0) return { qualifier: '', code: svc01 };
  return { qualifier: svc01.slice(0, idx), code: svc01.slice(idx + 1) };

export async function parse835(rawEdi: string): Promise<Parsed835> {
  if (!rawEdi || typeof rawEdi !== 'string') {
    throw new PayerConnectionError('835 parse failed: empty input');
  }
  if (!rawEdi.trimStart().startsWith('ISA')) {
    throw new PayerConnectionError('835 parse failed: missing ISA envelope');
  }

  let payer: PayerInfo = { name: '' };
  let payee: PayeeInfo = { name: '' };
  let payment: PaymentInfo = { amount: 0, method: '', checkNumber: '', checkDate: '' };
  const claims: ParsedClaim[] = [];
  const plbs: ProviderLevelAdjustment[] = [];

  let currentClaim: ParsedClaim | null = null;
  let currentLine: ParsedServiceLine | null = null;
  let segIndex = 0;

  try {
    // ACTUAL API CALL HERE — adapt to verified x12-parser surface from Step 2
    const stream = Readable.from([rawEdi]).pipe(new (await import('x12-parser')).X12Parser());

    for await (const rawSeg of stream as AsyncIterable<{ name: string; elements: string[] }>) {
      segIndex++;
      const seg = rawSeg;
      switch (seg.name) {
        case 'BPR':
          payment = {
            amount: Number(seg.elements[1] ?? '0'),
            method: seg.elements[3] ?? '',
            checkNumber: seg.elements[4] ?? '', // BPR05
            checkDate: parseDateYYMMDDorCCYYMMDD(seg.elements[15] ?? seg.elements[14] ?? ''),
          };
          break;
        case 'TRN':
          if (payment) payment.traceNumber = seg.elements[1] ?? '';
          break;
        case 'N1':
          if (seg.elements[0] === 'PR') payer = { name: seg.elements[1] ?? '' };
          if (seg.elements[0] === 'PE') {
            payee = { name: seg.elements[1] ?? '', npi: seg.elements[2] === 'XX' ? seg.elements[3] : undefined };
          }
          break;
        case 'CLP':
          if (currentClaim) claims.push(currentClaim);
          currentLine = null;
          currentClaim = {
            claimPmsId: seg.elements[0] ?? '',
            claimStatusCode: seg.elements[1] ?? '',
            billed: Number(seg.elements[2] ?? '0'),
            paid: Number(seg.elements[3] ?? '0'),
            patientResp: Number(seg.elements[4] ?? '0'),
            payerClaimControl: seg.elements[6] ?? undefined,
            serviceLines: [],
            claimLevelAdjustments: [],
          };
          break;
        case 'NM1':
          if (currentClaim && seg.elements[0] === 'QC') {
            currentClaim.patientLastName = seg.elements[2] ?? '';
            currentClaim.patientFirstName = seg.elements[3] ?? '';
            // element 8 = qualifier (MI = member ID), element 9 = the ID
            if (seg.elements[7] === 'MI') currentClaim.patientId = seg.elements[8];
          }
          break;
        case 'SVC': {
          if (!currentClaim) break;
          const composite = parseCompositeProcedureCode(seg.elements[0] ?? '');
          currentLine = {
            procedureCode: composite.code,
            procedureQualifier: composite.qualifier,
            fee: Number(seg.elements[1] ?? '0'),
            paid: Number(seg.elements[2] ?? '0'),
            units: seg.elements[4] ? Number(seg.elements[4]) : undefined,
            adjustments: [],
          };
          currentClaim.serviceLines.push(currentLine);
          break;
        }
        case 'DTM':
          if (currentLine && seg.elements[0] === '472') {
            currentLine.serviceDate = parseDateYYMMDDorCCYYMMDD(seg.elements[1] ?? '');
          }
          break;
        case 'CAS': {
          if (!currentClaim) break;
          // CAS can repeat groups: CAS*GROUP*CODE1*AMT1*Q1*CODE2*AMT2*Q2... in groups of 3
          for (let i = 1; i + 1 < seg.elements.length; i += 3) {
            const adj: ParsedAdjustment = {
              groupCode: (seg.elements[0] ?? '') as ParsedAdjustment['groupCode'],
              reasonCode: seg.elements[i] ?? '',
              amount: Number(seg.elements[i + 1] ?? '0'),
            };
            if (currentLine) currentLine.adjustments.push(adj);
            else currentClaim.claimLevelAdjustments.push(adj);
          }
          break;
        }
        case 'REF':
          if (currentLine && seg.elements[0] === '6R') {
            // line item control number — not tooth; tooth lives in different REF qualifier per dental companion guides
          }
          // Dental tooth numbers — Delta Dental companion guide; verify per fixture
          if (currentLine && (seg.elements[0] === 'TT' || seg.elements[0] === 'TQ')) {
            currentLine.toothNumber = seg.elements[1];
          }
          break;
        case 'PLB': {
          // PLB03 may be composite "WO:CLAIMNUM" or just "WO"; PLB04 is amount
          const reasonRaw = seg.elements[2] ?? '';
          const reason = reasonRaw.includes(':') ? reasonRaw.split(':')[0] : reasonRaw;
          plbs.push({
            providerId: seg.elements[0] ?? '',
            fiscalDate: parseDateYYMMDDorCCYYMMDD(seg.elements[1] ?? ''),
            reasonCode: reason,
            amount: Number(seg.elements[3] ?? '0'),
          });
          break;
        }
      }
    }
    if (currentClaim) claims.push(currentClaim);
  } catch (err) {
    // PHI scrub: never include raw EDI in error
    const safeMessage = err instanceof Error
      ? err.message.replace(PHI_SCRUB, '[REDACTED]')
      : 'unknown error';
    throw new PayerConnectionError(phiSafeMessage(null, segIndex) + ' — ' + safeMessage);
  }

  return { payer, payee, payment, claims, providerLevelAdjustments: plbs };
}

import type { Parsed835, ParsedClaim, ParsedServiceLine, ParsedAdjustment, ProviderLevelAdjustment } from './edi835-types';
import type { RawEOBDocument, RawEOBLineItem } from '../types';
import { PayerConnectionError } from '../types';

// Carrier reason-code → human-readable mapping for dental denial codes.
// Source: Delta Dental / DXC companion guides; expand as fixtures surface new codes.
const CARC_REASON: Record<string, string> = {
  '1': 'Deductible amount',
  '2': 'Coinsurance amount',
  '3': 'Co-payment amount',
  '4': 'Procedure code is inconsistent with the modifier used',
  '45': 'Charge exceeds maximum allowable',
  '50': 'Procedure not deemed medically necessary',
  '96': 'Non-covered charge(s)',
  '109': 'Claim not covered by this payer/contractor',
  '197': 'Precertification/authorization not obtained',
};

function mapDenialReason(code: string): string {
  return CARC_REASON[code] ?? `CARC ${code}`;
}

function sumAdjustmentsByGroup(adjustments: ParsedAdjustment[], group: ParsedAdjustment['groupCode']): number {
  return adjustments
    .filter((a) => a.groupCode === group)
    .reduce((sum, a) => sum + a.amount, 0);
}

function findFirstDenialAdjustment(adjustments: ParsedAdjustment[]): ParsedAdjustment | undefined {
  // CO (contractual) with non-zero amount and reason in known denial set, OR any line with paid=0 and an adjustment.
  // Rule: a denial is signaled by paid=0 AND a CO adjustment, OR by reason codes in the rejection range.
  const denialReasons = new Set(['4', '50', '96', '109', '197']);
  return adjustments.find((a) => denialReasons.has(a.reasonCode));
}

function mapServiceLine(claim: ParsedClaim, line: ParsedServiceLine): RawEOBLineItem {
  const co = sumAdjustmentsByGroup(line.adjustments, 'CO');
  const pr = sumAdjustmentsByGroup(line.adjustments, 'PR');
  const denial = findFirstDenialAdjustment(line.adjustments);

  // Allowed = billed - contractual obligation (CO). This is the dental 835 standard derivation.
  const allowed = Math.max(0, Math.round((line.fee - co) * 100) / 100);
  const adjustment = Math.round(co * 100) / 100;
  const paid = Math.round(line.paid * 100) / 100;
  const patientResponsibility = Math.round(pr * 100) / 100;

  return {
    patientFirstName: claim.patientFirstName ?? '',
    patientLastName: claim.patientLastName ?? '',
    patientPmsId: claim.patientId,
    procedureCode: line.procedureCode,
    toothNumber: line.toothNumber,
    serviceDate: line.serviceDate ?? '',
    fee: line.fee,
    allowed,
    paid,
    adjustment,
    patientResponsibility,
    denialCode: denial ? `${denial.groupCode}-${denial.reasonCode}` : undefined,
    denialReason: denial ? mapDenialReason(denial.reasonCode) : undefined,
    claimPmsId: claim.claimPmsId,
  };
}

function netPlbAmount(plbs: ProviderLevelAdjustment[]): number {
  return plbs.reduce((sum, p) => sum + p.amount, 0);
}

const RECONCILE_TOLERANCE_CENTS = 1; // 1 cent tolerance for floating-point math

export function mapParsed835ToRawEOB(parsed: Parsed835): RawEOBDocument {
  if (!parsed.payment.checkNumber) {
    throw new PayerConnectionError('835 mapping failed: missing BPR05 (check number)');
  }
  if (!parsed.payer.name) {
    throw new PayerConnectionError('835 mapping failed: missing N1*PR (payer name)');
  }

  // Reconciliation guard (RESEARCH.md anti-pattern §line 308)
  // BPR02 should equal sum(CLP04) + net(PLB). If it doesn't, flag rather than silently posting.
  const sumClpPaid = parsed.claims.reduce((sum, c) => sum + c.paid, 0);
  const netPlb = netPlbAmount(parsed.providerLevelAdjustments);
  const expected = Math.round((sumClpPaid + netPlb) * 100);
  const actual = Math.round(parsed.payment.amount * 100);
  if (Math.abs(expected - actual) > RECONCILE_TOLERANCE_CENTS) {
    throw new PayerConnectionError(
      `BPR/CLP reconciliation failed: BPR02=$${(actual / 100).toFixed(2)} but sum(CLP04)+net(PLB)=$${(expected / 100).toFixed(2)}`
    );
  }

  const lineItems: RawEOBLineItem[] = [];
  for (const claim of parsed.claims) {
    if (claim.serviceLines.length === 0) {
      // Claim with no service lines — synthesize a single-line entry from CLP totals
      lineItems.push({
        patientFirstName: claim.patientFirstName ?? '',
        patientLastName: claim.patientLastName ?? '',
        patientPmsId: claim.patientId,
        procedureCode: 'UNSPECIFIED',
        serviceDate: '',
        fee: claim.billed,
        allowed: claim.billed - sumAdjustmentsByGroup(claim.claimLevelAdjustments, 'CO'),
        paid: claim.paid,
        adjustment: sumAdjustmentsByGroup(claim.claimLevelAdjustments, 'CO'),
        patientResponsibility: claim.patientResp,
        claimPmsId: claim.claimPmsId,
      });
      continue;
    }
    for (const line of claim.serviceLines) {
      lineItems.push(mapServiceLine(claim, line));
    }
  }

  return {
    checkNumber: parsed.payment.checkNumber,
    checkDate: parsed.payment.checkDate,
    checkAmount: parsed.payment.amount,
    payerName: parsed.payer.name,
    receivedDate: new Date().toISOString().split('T')[0],
    source: 'edi_835',
    lineItems,
    rawData: {
      transactionTraceNumber: parsed.payment.traceNumber,
      payeeName: parsed.payee.name,
      payeeNpi: parsed.payee.npi,
      claimsCount: parsed.claims.length,
      providerLevelAdjustmentsCount: parsed.providerLevelAdjustments.length,
    },
  };
}
